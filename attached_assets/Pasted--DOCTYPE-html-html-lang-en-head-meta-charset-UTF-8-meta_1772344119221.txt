    <script>
        // ================= CONFIGURATION =================
        const STRIPE_PUBLIC_KEY = 'pk_live_your_stripe_key'; // REPLACE with your key
        const stripe = Stripe(STRIPE_PUBLIC_KEY);
        
        // State
        let currentQuestion = 1;
        let userAnswers = {
            name: '',
            problem: '',
            seriousness: '5'
        };
        
        let sessionTimer;
        let secondsLeft = 120; // 2 minutes
        let sessionActive = false;
        let sessionId = generateSessionId();
        let currentPlan = 'single';
        let upsellShown = false;
        
        // ================= INITIALIZATION =================
        function init() {
            updateLiveCounter();
            startSocialProofUpdates();
            checkForPaymentReturn();
        }
        
        function generateSessionId() {
            return 'sess_' + Math.random().toString(36).substring(2, 15) + Date.now().toString(36);
        }
        
        // ================= QUESTION FLOW =================
        function nextQuestion(num) {
            // Validate current question
            if (currentQuestion === 1 && !document.getElementById('q1').value.trim()) {
                alert('Dr. Trump needs your name!');
                return;
            }
            if (currentQuestion === 2 && !document.getElementById('q2').value.trim()) {
                alert('Tell Dr. Trump what\'s bothering you!');
                return;
            }
            
            // Save answers
            userAnswers.name = document.getElementById('q1').value.trim() || 'Friend';
            userAnswers.problem = document.getElementById('q2').value.trim() || 'life';
            userAnswers.seriousness = document.getElementById('q3').value;
            
            // Hide current, show next
            document.getElementById(`question${currentQuestion}`).style.display = 'none';
            document.getElementById(`question${num}`).style.display = 'block';
            
            currentQuestion = num;
            
            // Update progress
            const progress = ((num - 1) / 3) * 100;
            document.getElementById('progressFill').style.width = progress + '%';
            document.getElementById('questionCounter').textContent = `Question ${num - 1} of 3`;
        }
        
        // ================= THERAPY GENERATION =================
        async function startTherapy() {
            // Save final answers
            userAnswers.name = document.getElementById('q1').value.trim() || 'Friend';
            userAnswers.problem = document.getElementById('q2').value.trim() || 'life';
            userAnswers.seriousness = document.getElementById('q3').value;
            
            // Hide questions
            document.getElementById('question3').style.display = 'none';
            
            // Show session container
            document.getElementById('sessionContainer').style.display = 'block';
            
            // Generate therapy
            await generateTherapy();
            
            // Start timer
            startTimer();
            
            // Track session start
            trackEvent('session_start', userAnswers);
        }
        
        async function generateTherapy() {
            const response = await fetch('/api/generate-therapy', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(userAnswers)
            });
            
            const data = await response.json();
            document.getElementById('sessionContent').textContent = data.therapy;
            
            // Store for sharing
            sessionStorage.setItem('lastTherapy', data.therapy);
            sessionStorage.setItem('userName', userAnswers.name);
        }
        
        // ================= TIMER =================
        function startTimer() {
            sessionActive = true;
            secondsLeft = 120;
            
            sessionTimer = setInterval(() => {
                secondsLeft--;
                
                // Update display
                const mins = Math.floor(secondsLeft / 60);
                const secs = secondsLeft % 60;
                document.getElementById('timerDisplay').textContent = `${mins}:${secs.toString().padStart(2, '0')}`;
                
                // Warning at 30 seconds
                if (secondsLeft === 30) {
                    document.getElementById('timerDisplay').classList.add('warning');
                    
                    // Show upsell if not shown yet
                    if (!upsellShown) {
                        showUpsell();
                        upsellShown = true;
                    }
                }
                
                // End session
                if (secondsLeft <= 0) {
                    endSession();
                }
                
            }, 1000);
        }
        
        function endSession() {
            clearInterval(sessionTimer);
            sessionActive = false;
            
            // Show share section
            generateShareSnippets();
            document.getElementById('shareSection').style.display = 'block';
            
            // Show upgrade card
            document.getElementById('upgradeCard').style.display = 'block';
            
            // Track completion
            trackEvent('session_complete', { duration: 120 - secondsLeft });
        }
        
        // ================= SHARE MECHANICS =================
        function generateShareSnippets() {
            const therapy = sessionStorage.getItem('lastTherapy') || document.getElementById('sessionContent').textContent;
            const name = sessionStorage.getItem('userName') || userAnswers.name;
            
            // Break into sentences
            const sentences = therapy.split('. ').filter(s => s.length > 20);
            const snippets = sentences.slice(0, 3);
            
            let html = '';
            snippets.forEach((snippet, i) => {
                html += `
                    <div class="snippet-card" onclick="shareSnippet(${i}, '${encodeURIComponent(snippet)}')">
                        <div class="snippet-text">"${snippet.substring(0, 80)}..."</div>
                        <div class="snippet-share">🔁 Click to share</div>
                    </div>
                `;
            });
            
            document.getElementById('snippetGrid').innerHTML = html;
        }
        
        function shareSnippet(index, snippet) {
            const name = sessionStorage.getItem('userName') || userAnswers.name;
            const text = encodeURIComponent(`Trump told me: "${decodeURIComponent(snippet)}" Get your therapy at trumpbot.rip 🔥`);
