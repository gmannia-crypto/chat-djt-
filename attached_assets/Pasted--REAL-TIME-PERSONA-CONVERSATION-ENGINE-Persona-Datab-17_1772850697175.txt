        const state = {};
        this.activePersonas.forEach(id => {
            state[id] = {
                anger: 20,
                happiness: 50,
                engagement: 50,
                lastSpoke: null,
                topics: []
            };
        });
        return state;
    }

    startConversation() {
        // Trump starts
        setTimeout(() => {
            this.speak('trump', "You know, Netanyahu is my friend. A great friend. The best prime minister Israel ever had, believe me!");
        }, 1000);
        
        // Start the conversation loop
        this.conversationLoop();
    }

    conversationLoop() {
        setInterval(() => {
            if (!this.listening) return;
            
            // Analyze last few messages and decide who should speak next
            this.decideNextSpeaker();
        }, 3000);
    }

    decideNextSpeaker() {
        const lastMessages = this.conversationHistory.slice(-3);
        if (lastMessages.length === 0) return;
        
        const lastSpeaker = lastMessages[lastMessages.length - 1].speaker;
        const lastText = lastMessages[lastMessages.length - 1].text;
        
        // Check if anyone should respond
        this.activePersonas.forEach(personaId => {
            if (personaId === lastSpeaker) return; // Don't respond to self
            
            // Calculate probability of response based on relationship and content
            const probability = this.calculateResponseProbability(personaId, lastSpeaker, lastText);
            
            if (Math.random() * 100 < probability) {
                // Persona wants to respond
                this.queueResponse(personaId, lastSpeaker, lastText);
            }
        });
    }

    calculateResponseProbability(listenerId, speakerId, text) {
        const listener = PERSONAS[listenerId];
        const relationship = listener.relationships[speakerId] || { sentiment: 50 };
        const emotionalState = this.emotionalState[listenerId];
        
        // Base probability
        let probability = 30;
        
        // Adjust based on relationship
        probability += (relationship.sentiment - 50) * 0.3;
        
        // Adjust based on engagement
        probability += (emotionalState.engagement - 50) * 0.2;
        
        // Check for trigger words
        const words = text.toLowerCase().split(' ');
        words.forEach(word => {
            if (listener.triggerWords.positive.includes(word)) {
                probability += 10;
            }
            if (listener.triggerWords.negative.includes(word)) {
                probability += 15; // More likely to respond to negative triggers
            }
        });
        
        // Cap probability
        return Math.min(80, Math.max(10, probability));
    }

    queueResponse(responderId, toSpeakerId, text) {
        // Don't queue if already in queue
        if (this.speakingQueue.some(item => item.responderId === responderId)) return;
        
        this.speakingQueue.push({
            responderId,
            toSpeakerId,
            text,
            timestamp: Date.now()
        });
        
        // Process queue
        this.processQueue();
    }

    async processQueue() {
        if (this.speakingQueue.length === 0 || this.currentSpeaker) return;
        
        // Sort queue by priority (based on engagement)
        this.speakingQueue.sort((a, b) => {
            return this.emotionalState[b.responderId].engagement - this.emotionalState[a.responderId].engagement;
        });
        
        const next = this.speakingQueue.shift();
        await this.generateResponse(next.responderId, next.toSpeakerId, next.text);
    }

    async generateResponse(responderId, toSpeakerId, lastText) {
        this.currentSpeaker = responderId;
        
        // Show speaking indicator
        document.querySelectorAll('.speaking-indicator').forEach(el => el.style.display = 'none');
        document.getElementById(`speaking-${responderId}`).style.display = 'block';
        
        // Generate contextual response based on persona
        const response = this.createContextualResponse(responderId, toSpeakerId, lastText);
        
        // Add to conversation
        this.addToConversation(responderId, response);
        
        // Update emotional states
        this.updateEmotionalStates(responderId, toSpeakerId, response);
        
        // Speak if voice enabled
        if (this.voiceEnabled) {
            this.speak(responderId, response);
        }
        
        // Clear speaking indicator after delay
        setTimeout(() => {
            document.getElementById(`speaking-${responderId}`).style.display = 'none';
            this.currentSpeaker = null;
            
            // Check if others should respond to this response
            setTimeout(() => {
                this.decideNextSpeaker();
            }, 1000);
        }, 3000 + (response.length * 50)); // Time based on response length
    }

    extractTopics(text) {
        const topics = [];
        const topicKeywords = {
            'israel': ['israel', 'jerusalem', 'gaza', 'west bank', 'palestine'],
            'iran': ['iran', 'nuclear', 'tehran', 'ayatollah'],
            'biden': ['biden', 'sleepy joe', 'joe', 'president biden'],
            'economy': ['economy', 'inflation', 'jobs', 'stock market', 'recession'],
            'media': ['media', 'fake news', 'press', 'cnn', 'msnbc', 'fox'],
            'britain': ['britain', 'uk', 'london', 'british', 'england']
        };
        
        const lowerText = text.toLowerCase();
        Object.entries(topicKeywords).forEach(([topic, keywords]) => {
            if (keywords.some(keyword => lowerText.includes(keyword))) {
                topics.push(topic);
            }
        });
        
        return topics;
    }

    updateEmotionalStates(responderId, toSpeakerId, response) {
        // Update responder's engagement
        this.emotionalState[responderId].engagement = Math.min(100, this.emotionalState[responderId].engagement + 10);
        this.emotionalState[responderId].lastSpoke = Date.now();
        
        // Update based on relationship
        const relationship = PERSONAS[responderId].relationships[toSpeakerId] || { sentiment: 50 };
        
        if (relationship.sentiment > 70) {
            this.emotionalState[responderId].happiness = Math.min(100, this.emotionalState[responderId].happiness + 5);
            this.emotionalState[responderId].anger = Math.max(0, this.emotionalState[responderId].anger - 3);
        } else if (relationship.sentiment < 30) {
            this.emotionalState[responderId].happiness = Math.max(0, this.emotionalState[responderId].happiness - 3);
            this.emotionalState[responderId].anger = Math.min(100, this.emotionalState[responderId].anger + 8);
        }
        
        // Update other personas' engagement (they're listening)
        this.activePersonas.forEach(id => {
            if (id !== responderId) {
                this.emotionalState[id].engagement = Math.min(100, this.emotionalState[id].engagement + 2);
            }
        });
        
        // Update UI
        this.updateEmotionalUI();
    }

    updateEmotionalUI() {
        const grid = document.getElementById('emotionalGrid');
        if (grid) {
            grid.innerHTML = this.renderEmotionalState();
        }
        
        this.activePersonas.forEach(id => {
            const emotionEl = document.getElementById(`emotion-${id}`);
            if (emotionEl) {
                emotionEl.innerHTML = `
                    <div class="emotion-bar anger" style="width: ${this.emotionalState[id].anger}%"></div>
                    <div class="emotion-bar happy" style="width: ${this.emotionalState[id].happiness}%"></div>
                `;
            }
        });
    }

    addToConversation(speakerId, text) {
        const speaker = PERSONAS[speakerId];
        const timestamp = new Date().toLocaleTimeString();
        
        this.conversationHistory.push({
            speaker: speakerId,
            text: text,
            timestamp: timestamp
        });
        
        const stream = document.getElementById('streamMessages');
        const message = document.createElement('div');
        message.className = 'stream-message';
        message.style.borderLeftColor = speaker.color;
        message.innerHTML = `
            <div class="message-header">
                <span class="message-avatar">${speaker.avatar}</span>
                <span class="message-speaker" style="color: ${speaker.color};">${speaker.name}</span>
                <span class="message-time">${timestamp}</span>
            </div>
            <div class="message-text">"${text}"</div>
        `;
        
        stream.appendChild(message);
        stream.scrollTop = stream.scrollHeight;
        
        // Keep only last 50 messages
        while (stream.children.length > 50) {
            stream.removeChild(stream.children[1]);
        }
    }

            alert("All personas are already in the conversation!");
            return;

        const newPersona = available[0];
        this.activePersonas.push(newPersona);
        this.emotionalState[newPersona] = {
            anger: 30,
            happiness: 50,
            engagement: 50,
            lastSpoke: null,
            topics: []

        // Re-render

        // New persona joins conversation
        setTimeout(() => {
            this.addToConversation(newPersona, `Hello everyone. Important to join this conversation.`);
        }, 1000);
    }

    }
