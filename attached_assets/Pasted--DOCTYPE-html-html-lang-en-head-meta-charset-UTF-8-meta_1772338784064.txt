    <script>
        // ================= CONFIGURATION =================
        const STRIPE_PUBLIC_KEY = 'pk_live_your_stripe_key'; // Replace with your Stripe key
        const stripe = Stripe(STRIPE_PUBLIC_KEY);
        
        // State management
        let currentPlan = 'single';
        let userName = '';
        let userProblem = '';
        let freeSessions = parseInt(localStorage.getItem('freeSessions') || '3');
        let lastSessionTime = localStorage.getItem('lastSessionTime') || '0';
        let sessionCount = parseInt(localStorage.getItem('sessionCount') || '0');
        
        // ================= INITIALIZATION =================
        function init() {
            updateFreeSessions();
            checkLastSession();
            startFOMOTimer();
            setTimeout(showUrgencyBadge, 30000); // Show after 30 seconds
            updateSocialProof();
        }
        
        // ================= FREE SESSION MANAGEMENT =================
        function updateFreeSessions() {
            const today = new Date().toDateString();
            const lastReset = localStorage.getItem('lastReset');
            
            if (lastReset !== today) {
                // Reset free sessions for new day
                freeSessions = 3;
                localStorage.setItem('freeSessions', '3');
                localStorage.setItem('lastReset', today);
            }
            
            document.getElementById('freeSessions').textContent = freeSessions;
            
            if (freeSessions <= 0) {
                document.getElementById('sessionCounter').style.display = 'none';
                document.getElementById('premiumUpsell').style.display = 'block';
                document.getElementById('payButton').textContent = '🔥 UNLOCK NOW - $2.99 🔥';
            } else {
                document.getElementById('payButton').textContent = `🎁 USE FREE SESSION (${freeSessions} left) 🎁`;
            }
        }
        
        function checkLastSession() {
            const now = Date.now();
            const last = parseInt(lastSessionTime);
            const hoursSinceLast = (now - last) / (1000 * 60 * 60);
            
            if (hoursSinceLast < 24 && freeSessions < 3) {
                // Sessions haven't reset yet
                document.getElementById('freeSessions').textContent = freeSessions;
            }
        }
        
        // ================= THERAPY GENERATION =================
        async function generateTherapy(name, problem, seriousness) {
            // Trump therapy response templates
            const responses = [
                `"${name}, let me tell you something about ${problem}. I've had the same problem. The BEST people have this problem. You know what I did? I won. I won so much. And you're going to win too. Believe me."`,
                
                `"${problem}? That's nothing. I've faced witch hunts, fake news, the whole thing. And I came out stronger. ${name}, you're a fighter. You have Trump DNA. Now go out there and be tremendous."`,
                
                `"I'm looking at your situation, ${name}. ${problem} is serious. Very serious. But here's the thing - you're smarter than them. You're stronger. On a scale of 1-10, this problem is a ${seriousness}. Your ability to fix it? 100. You're welcome."`,
                
                `"${name}, I've employed thousands of people. The best people. And you know what makes someone great? How they handle ${problem}. And you? You're handling it beautifully. Could be better. But beautiful."`,
                
                `"The fake news would tell you ${problem} is your fault. Wrong! It's their fault. Everything is their fault. ${name}, stop listening to the haters. You're doing great. Tremendous, even."`
            ];
            
            // Add comment-based responses
            if (problem.toLowerCase().includes('work') || problem.toLowerCase().includes('job')) {
                responses.push(`"Work problems? ${name}, you're probably underpaid. Very underpaid. I know salaries. I pay the best salaries. Ask for a raise. If they say no, tell them Trump sent you. Works every time."`);
            }
            if (problem.toLowerCase().includes('love') || problem.toLowerCase().includes('relationship')) {
                responses.push(`"${name}, love is complicated. I've been married, what, three times? Great marriages. The best. But ${problem}? You'll find someone. Someone tremendous. And if not, you've still got me. I'm always here."`);
            }
            if (problem.toLowerCase().includes('money') || problem.toLowerCase().includes('broke')) {
                responses.push(`"Money problems? ${name}, I've been broke before. Many times. And I came back bigger, better, richer. You will too. Invest in walls. Walls always win."`);
            }
            
            // Simulate AI delay
            await new Promise(resolve => setTimeout(resolve, 1500));
            
            return responses[Math.floor(Math.random() * responses.length)];
        }
        
        // ================= PAYMENT HANDLING =================
        async function handlePayment() {
            const name = document.getElementById('userName').value.trim();
            const problem = document.getElementById('userProblem').value.trim();
            const seriousness = document.getElementById('seriousness').value;
            
            if (!name || !problem) {
                alert('Please tell Dr. Trump your name and what\'s bothering you.');
                return;
            }
            
            userName = name;
            userProblem = problem;
            
            // Check if free session available
            if (freeSessions > 0) {
                // Use free session
                await useFreeSession();
            } else {
                // Process payment
                await processPayment();
            }
        }
        
        async function useFreeSession() {
            const button = document.getElementById('payButton');
            button.classList.add('loading');
            button.disabled = true;
            
            try {
                // Generate therapy
                const therapy = await generateTherapy(userName, userProblem, document.getElementById('seriousness').value);
                
                // Update free sessions
                freeSessions--;
                localStorage.setItem('freeSessions', freeSessions.toString());
                localStorage.setItem('lastSessionTime', Date.now().toString());
                sessionCount++;
                localStorage.setItem('sessionCount', sessionCount.toString());
                
                // Show result
                showTherapyResult(therapy);
                
                // Track for viral analytics
                trackSession('free');
                
            } catch (error) {
                alert('Session failed. Please try again.');
            } finally {
                button.classList.remove('loading');
                button.disabled = false;
                updateFreeSessions();
            }
        }
        
        async function processPayment() {
            const button = document.getElementById('payButton');
            button.classList.add('loading');
            button.disabled = true;
            
            try {
                // Create Stripe checkout session
                const response = await fetch('/api/create-checkout', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        plan: currentPlan,
                        successUrl: window.location.href + '?session=success',
                        cancelUrl: window.location.href,
                        metadata: {
                            name: userName,
                            problem: userProblem,
                            seriousness: document.getElementById('seriousness').value
                        }
                    })
                });
                
                const session = await response.json();
                
                // Redirect to Stripe
                const result = await stripe.redirectToCheckout({
                    sessionId: session.id
                });
                
                if (result.error) {
                    alert(result.error.message);
                }
                
            } catch (error) {
                alert('Payment failed. Please try again.');
                button.classList.remove('loading');
                button.disabled = false;
            }
        }
        
        // ================= SHOW RESULTS =================
        async function showTherapyResult(therapy) {
            document.getElementById('therapyText').textContent = `"${therapy}"`;
            document.getElementById('sessionResult').style.display = 'block';
            
            // Scroll to result
            document.getElementById('sessionResult').scrollIntoView({ behavior: 'smooth' });
            
            // Update social proof
            updateSocialProof();
            
            // Trigger viral share tracking
            setupShareTracking(therapy);
        }
        
        // ================= VIRAL SHARE MECHANICS =================
        function setupShareTracking(therapy) {
            // Store for sharing
            sessionStorage.setItem('lastTherapy', therapy);
            sessionStorage.setItem('lastUserName', userName);
        }
        
        function shareTo(platform) {
            const therapy = sessionStorage.getItem('lastTherapy') || document.getElementById('therapyText').textContent;
            const name = sessionStorage.getItem('lastUserName') || userName;
            
            // Truncate for sharing
            const preview = therapy.substring(0, 100) + '...';
            
            // Create share text

            switch(platform) {
                case 'twitter':

                    return;
            }
            
            if (shareUrl) {
                window.open(shareUrl, '_blank', 'width=600,height=400');
                
                // Track share
                fetch('/api/track-share', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        platform: platform,
                        sessionId: sessionCount
                    })
                });
                
                // Show thank you message
                showShareThankYou();
            }

        function startFOMOTimer() {
            // Random timer between 2-7 minutes
            const minutes = Math.floor(Math.random() * 5) + 2;
            const seconds = minutes * 60;
            let timeLeft = seconds;
            
            const timer = document.querySelector('.timer');
            
            const interval = setInterval(() => {
                timeLeft--;
                
                if (timeLeft <= 0) {
                    // Reset with new random time
                    timeLeft = (Math.floor(Math.random() * 5) + 2) * 60;
                    
                    // Change message occasionally
                    const messages = [
                        '⏰ Special offer expiring soon!',
                        '⚡ 127 people looking at this right now',
                        '🔥 Only 3 free sessions left today',
                        '🎁 Your free session expires in '
                    ];
                    
                    const fomoText = document.querySelector('.fomo-timer');
                    const randomMsg = messages[Math.floor(Math.random() * messages.length)];
                    
                    if (randomMsg.includes('expires')) {
                        fomoText.innerHTML = `⏰ ${randomMsg} <span class="timer">${formatTime(timeLeft)}</span>`;
                    } else {
                        fomoText.innerHTML = `⏰ ${randomMsg}`;
                    }
                }
                
                const timerElement = document.querySelector('.timer');
                if (timerElement) {
                    timerElement.textContent = formatTime(timeLeft);
                }
                
            }, 1000);
        }
        
        function formatTime(seconds) {
            const mins = Math.floor(seconds / 60);
            const secs = seconds % 60;
            return `${mins}:${secs.toString().padStart(2, '0')}`;
        }
