    attachEvents() {
      // Tab switching
      const tabs = this.container.querySelectorAll('.tab-btn');
      tabs.forEach(tab => {
        tab.addEventListener('click', (e) => {
          const tabName = e.target.dataset.tab;
          this.switchTab(tabName);
        });
      });
      
      // Track affiliate clicks
      const links = this.container.querySelectorAll('a[data-affiliate]');
      links.forEach(link => {
        link.addEventListener('click', (e) => {
          const affiliate = link.dataset.affiliate;
          this.trackAffiliateClick(affiliate);
        });
      });
    }
    
    switchTab(tabName) {
      this.activeTab = tabName;
      
      // Update tab buttons
      const tabs = this.container.querySelectorAll('.tab-btn');
      tabs.forEach(tab => {
        tab.classList.toggle('active', tab.dataset.tab === tabName);
      });
      
      // Update content
      const contentMap = {
        live: this.renderLiveTab(),
        commentary: this.renderCommentaryTab(),
        gear: this.renderGearTab(),
        betting: this.renderBettingTab()
      };
      
      const contentDiv = this.container.querySelector('#mediaTabContent');
      contentDiv.innerHTML = contentMap[tabName];
      
      // Re-attach click handlers for new links
      const newLinks = contentDiv.querySelectorAll('a[data-affiliate]');
      newLinks.forEach(link => {
        link.addEventListener('click', (e) => {
          const affiliate = link.dataset.affiliate;
          this.trackAffiliateClick(affiliate);
        });
      });
    }
    
    trackAffiliateClick(affiliateId) {
      console.log(`💰 Affiliate click: ${affiliateId}`);
      
      // Send to your analytics
      fetch('/api/track-affiliate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          affiliate: affiliateId,
          timestamp: Date.now(),
          page: 'sports-media-hub'
        })
      }).catch(() => {});
      
      // Optional: show notification
      if (window.SoundManager) {
        SoundManager.play('click');
      }
    }

      window.sportsMediaHub = new SportsMediaHub(containerId);
    }
  };
  
  // Update showFeature function to handle this new feature
  const originalShowFeature = window.showFeature;
  window.showFeature = function(feature) {
    if (originalShowFeature) {
      originalShowFeature(feature);
    }
    
    if (feature === 'sports-media') {
      const container = document.getElementById('sportsMediaHub');
      if (container) {
        container.style.display = 'block';
        initSportsMediaHub('sportsMediaHub');
      }
    }
