    // Start a new debate
    startDebate() {
        const assetKey = document.getElementById('assetSelect').value;
        const persona1Key = document.getElementById('persona1Select').value;
        const persona2Key = document.getElementById('persona2Select').value;
        
        const asset = FINANCIAL_TOPICS[assetKey];
        const persona1 = FINANCIAL_PERSONAS[persona1Key];
        const persona2 = FINANCIAL_PERSONAS[persona2Key];
        
        this.currentDebate = {
            id: Date.now(),
            asset: asset,
            persona1: persona1,
            persona2: persona2,
            votes: { [persona1Key]: 0, [persona2Key]: 0 },
            timestamp: new Date().toISOString()
        };
        
        this.renderDebate(asset, persona1, persona2);
        this.trackEvent('debate_started', { asset: assetKey, p1: persona1Key, p2: persona2Key });
    }

    // Handle voting
    vote(personaId) {
        if (!this.currentDebate) return;
        
        const p1Id = this.currentDebate.persona1.id;
        const p2Id = this.currentDebate.persona2.id;
        
        if (personaId === p1Id) {
            this.currentDebate.votes[p1Id]++;
        } else if (personaId === p2Id) {
            this.currentDebate.votes[p2Id]++;
        }
        
        this.updateVoteDisplay();
        this.trackEvent('vote_cast', { 
            debateId: this.currentDebate.id,
            votedFor: personaId 
        });
        
        // Show popup encouragement
        this.showVoteEncouragement();
    }
    
    // Update vote display
    updateVoteDisplay() {
        if (!this.currentDebate) return;
        
        const p1 = this.currentDebate.persona1;
        const p2 = this.currentDebate.persona2;
        const votes1 = this.currentDebate.votes[p1.id] || 0;
        const votes2 = this.currentDebate.votes[p2.id] || 0;
        const total = votes1 + votes2 || 1; // Avoid division by zero
        
        const p1Percent = (votes1 / total) * 100;
        const p2Percent = (votes2 / total) * 100;
        
        document.getElementById('p1Progress').style.width = p1Percent + '%';
        document.getElementById('p2Progress').style.width = p2Percent + '%';
        document.getElementById('p1Votes').textContent = votes1;
        document.getElementById('p2Votes').textContent = votes2;
        
        // Store in debate history
        this.currentDebate.finalVotes = { p1: votes1, p2: votes2 };
    }
    
    // Share debate to social media
    shareDebate(platform) {
        if (!this.currentDebate) return;
        
        const p1 = this.currentDebate.persona1;
        const p2 = this.currentDebate.persona2;
        const asset = this.currentDebate.asset;
        const votes1 = this.currentDebate.votes[p1.id] || 0;
        const votes2 = this.currentDebate.votes[p2.id] || 0;
        
        const shareText = encodeURIComponent(
            `💰 FINANCIAL FACEOFF: ${p1.name} vs ${p2.name} on ${asset.name}!\n` +
            `Current vote: ${votes1}-${votes2}\n` +
            `${p1.advice[asset.category] || p1.advice.default}\n\n` +
            `Who's right? Vote and debate at trumpbot.rip!`
        );
        
        const shareUrl = encodeURIComponent('https://trumpbot.rip/financial-faceoff');
        
        switch(platform) {
            case 'twitter':
                window.open(`https://twitter.com/intent/tweet?text=${shareText}&url=${shareUrl}`, '_blank');
                break;
            case 'facebook':
                window.open(`https://www.facebook.com/sharer/sharer.php?u=${shareUrl}&quote=${shareText}`, '_blank');
                break;
            case 'copy':
                navigator.clipboard.writeText(decodeURIComponent(shareText) + ' https://trumpbot.rip/financial-faceoff');
                alert('✅ Debate copied to clipboard!');
                break;
        }
        
        this.trackEvent('debate_shared', { platform });
    }
    
    // Show vote encouragement popup
    showVoteEncouragement() {
        const encouragement = [
            "🔥 Keep voting! The winner gets bragging rights!",
            "📊 Your vote matters in this financial showdown!",
            "⚡ Only 30 seconds left to vote!",
            "🎯 Who do YOU trust with your money?"
        ];
        
        const random = encouragement[Math.floor(Math.random() * encouragement.length)];
        
        // Create toast notification
        const toast = document.createElement('div');
        toast.className = 'vote-toast';
        toast.innerHTML = random;
        toast.style.cssText = `
            position: fixed;
            bottom: 20px;
            right: 20px;
            background: #ff4d4d;
            color: white;
            padding: 15px 25px;
            border-radius: 50px;
            font-weight: bold;
            animation: slideIn 0.3s;
            z-index: 10000;
        `;
        
        document.body.appendChild(toast);
        setTimeout(() => toast.remove(), 3000);
    }
    
    // Render affiliate link if available
    renderAffiliateLink(persona, asset) {
        // Find matching affiliate link
        let link = null;
        let text = '';
        
        if (asset.category === 'crypto' && persona.affiliate.coinbase) {
            link = persona.affiliate.coinbase;
            text = `💰 Buy ${asset.name} on Coinbase`;
        } else if (asset.category === 'crypto' && persona.affiliate.binance) {
            link = persona.affiliate.binance;
            text = `💰 Trade ${asset.name} on Binance`;
        } else if (asset.category === 'stock' && persona.affiliate.vanguard) {
            link = persona.affiliate.vanguard;
            text = `📈 Invest in ${asset.name} via Vanguard`;
        } else if (persona.affiliate.vanguard) {
            link = persona.affiliate.vanguard;
            text = `📊 Start investing with Vanguard`;
        }
        
        if (link) {
            return `
                <a href="${link}" target="_blank" class="affiliate-link" onclick="faceoff.trackAffiliate('${persona.id}')">
                    ${text} <span class="affiliate-badge">Sponsored</span>
                </a>
            `;
        }
        return '';
    }
    
    // Track affiliate clicks
    trackAffiliate(personaId) {
        this.trackEvent('affiliate_click', { 
            persona: personaId,
            debateId: this.currentDebate?.id 
        });
    }
    
    // Get persona title
    getTitle(personaId) {
        const titles = {
            trump: '45th President • Real Estate Mogul',
            buffett: 'CEO Berkshire Hathaway • Oracle of Omaha',
            musk: 'CEO Tesla/SpaceX • Chief Twit',
            suze: 'Financial Advisor • TV Host • Author',
            dave: 'Radio Host • Financial Author',
            grandma: 'Lifetime Saver • Cookie Baker',
            robot: 'AI Financial Algorithm v2.0'
        };
        return titles[personaId] || 'Financial Expert';
    }
    
    // Track events for analytics
    trackEvent(eventName, data) {
        console.log(`📊 FINANCIAL FACEOFF EVENT: ${eventName}`, data);
        
        // Send to analytics (implement your analytics here)
        fetch('/api/track-financial-event', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                event: eventName,
                data: data,
                timestamp: Date.now(),
                sessionId: this.getSessionId()
            })
        }).catch(() => {});
    }
    
    // Get or create session ID
    getSessionId() {
        let sessionId = sessionStorage.getItem('financialSessionId');
        if (!sessionId) {
            sessionId = 'fin_' + Math.random().toString(36).substring(2, 15);
            sessionStorage.setItem('financialSessionId', sessionId);
        }
        return sessionId;
    }
