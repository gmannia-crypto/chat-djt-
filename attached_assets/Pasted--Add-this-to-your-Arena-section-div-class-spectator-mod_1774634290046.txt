<script>
let viewerCount = 1247;
let reactionInterval;

function updateViewerCount() {
  viewerCount += Math.floor(Math.random() * 10) - 3;
  viewerCount = Math.max(500, viewerCount);
  document.getElementById('viewerCount').innerHTML = viewerCount.toLocaleString() + ' watching';
}

function sendReaction(emoji) {
  addChatMessage(`Anonymous: ${emoji}`, 'reaction');
}

function sendChat() {
  const input = document.getElementById('chatInput');
  const message = input.value.trim();
  if (message) {
    addChatMessage(`You: ${message}`, 'user');
    input.value = '';
    // Simulate other users reacting
    setTimeout(() => {
      const responses = ['🔥', '😂', '👏', 'Interesting take!', 'Facts!'];
      const random = responses[Math.floor(Math.random() * responses.length)];
      addChatMessage(`@User${Math.floor(Math.random() * 100)}: ${random}`);
    }, 1000);
  }
}

function addChatMessage(message, type = '') {
  const chat = document.getElementById('liveChat');
  const msgDiv = document.createElement('div');
  msgDiv.className = 'chat-message';
  if (type === 'reaction') msgDiv.style.color = '#ffaa00';
  if (type === 'user') msgDiv.style.fontWeight = 'bold';
  msgDiv.innerHTML = message;
  chat.appendChild(msgDiv);
  chat.scrollTop = chat.scrollHeight;
  // Keep only last 50 messages
  while (chat.children.length > 50) chat.removeChild(chat.firstChild);
}

// Simulate live debate content
function updateDebateStream() {
  const debates = [
    '<div class="debate-message"><strong>Trump:</strong> "Biden has destroyed this country! Believe me!"</div>',
    '<div class="debate-message"><strong>Biden:</strong> "That\'s malarkey! Come on, man."</div>',
    '<div class="debate-message"><strong>Ruckus:</strong> "TELL EM, TRUMP!"</div>'
  ];
  const stream = document.getElementById('debateStream');
  const newMsg = debates[Math.floor(Math.random() * debates.length)];
  stream.innerHTML += newMsg;
  while (stream.children.length > 5) stream.removeChild(stream.firstChild);
}

setInterval(updateViewerCount, 10000);
setInterval(updateDebateStream, 15000);
</script>
