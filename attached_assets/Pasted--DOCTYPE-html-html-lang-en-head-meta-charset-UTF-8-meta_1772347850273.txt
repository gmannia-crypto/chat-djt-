            <p style="color: #ccc;">Get 3 MORE MINUTES with your therapist</p>
            <div class="upsell-price">$1.99</div>
            <button class="upsell-btn" onclick="buyMoreTime()">ADD 3 MINUTES</button>

    <script>
        // ================= CONFIG =================
        const STRIPE_PUBLIC_KEY = 'pk_live_your_stripe_key'; // REPLACE WITH YOUR KEY
        const stripe = Stripe(STRIPE_PUBLIC_KEY);
        
        // State
        let selectedVoice = 'trump';
        let currentQuestion = 1;
        let userAnswers = { name: '', problem: '', seriousness: '5' };
        let sessionTimer;
        let secondsLeft = 120;
        let sessionActive = false;
        let sessionId = generateSessionId();
        let currentPlan = 'single';
        let upsellShown = false;
        let followUpIndex = 0;
        let followUpActive = false;
        let currentFollowUpQuestion = '';
        let exchanges = [];
        
        // Voice configurations
        const voiceConfig = {
            trump: {
                name: 'DR. TRUMP',
                avatar: '🗣️',
                color: '#ff4d4d',
                greeting: '"Lie down. Tell me everything. I\'m listening..."',
                q1: '"What\'s your name? I need to know who I\'m talking to."',
                q2: '"What\'s bothering you? Be specific. I deal with big problems."',
                q3: '"On a scale of 1-10, how serious is this? Be honest."'
            },
            sophia: {
                name: 'DR. SOPHIA',
                avatar: '👩‍⚕️',
                color: '#ff99cc',
                greeting: '"Welcome, dear. Take your time. I\'m here to listen without judgment."',
                q1: '"What\'s your name? I\'d love to get to know you."',
                q2: '"What\'s on your heart today? Share whatever feels right."',
                q3: '"How heavy does this feel for you, on a scale of 1-10?"'
            },
            james: {
                name: 'DR. JAMES',
                avatar: '👨‍⚕️',
                color: '#4d4dff',
                greeting: '"Hello. I\'m Dr. James. Let\'s work through this together."',
                q1: '"Your name? Let\'s start there."',
                q2: '"What brings you here today? Be as specific as you like."',
                q3: '"Rate the severity 1-10. This helps me understand."'
            }
        };
        
        // ================= INIT =================
        function init() {
            selectVoice('trump'); // Default
            updateLiveCounter();
            startSocialProofUpdates();
            checkForPaymentReturn();
        }
        
        function generateSessionId() {
            return 'sess_' + Math.random().toString(36).substring(2, 15);
        }
        
        // ================= VOICE SELECTION =================
        function selectVoice(voice) {
            selectedVoice = voice;
            
            // Update UI
            document.querySelectorAll('.voice-card').forEach(card => {
                card.classList.remove('selected');
                card.style.borderColor = '#333';
            });
            
            document.getElementById(`voice-${voice}`).classList.add('selected');
            document.getElementById(`voice-${voice}`).style.borderColor = voiceConfig[voice].color;
            
            // Update greeting and questions
            document.getElementById('greetingQuote').textContent = voiceConfig[voice].greeting;
            document.getElementById('q1Text').textContent = voiceConfig[voice].q1;
            document.getElementById('q2Text').textContent = voiceConfig[voice].q2;
            document.getElementById('q3Text').textContent = voiceConfig[voice].q3;
            
            // Update timer color
            document.getElementById('timerDisplay').className = `timer-display ${voice}`;
            
            trackEvent('voice_selected', { voice });
        }
        
        // ================= QUESTION FLOW =================
        function nextQuestion(num) {
            if (currentQuestion === 1 && !document.getElementById('q1').value.trim()) {
                alert('Please enter your name');
                return;
            }
            if (currentQuestion === 2 && !document.getElementById('q2').value.trim()) {
                alert('Please tell us what\'s bothering you');
                return;
            }
            
            document.getElementById(`question${currentQuestion}`).style.display = 'none';
            document.getElementById(`question${num}`).style.display = 'block';
            
            currentQuestion = num;
            const progress = ((num - 1) / 3) * 100;
            document.getElementById('progressFill').style.width = progress + '%';
            document.getElementById('questionCounter').textContent = `Question ${num - 1} of 3`;
        }
        
        // ================= START THERAPY =================
        async function startTherapy() {
            userAnswers = {
                name: document.getElementById('q1').value.trim() || 'Friend',
                problem: document.getElementById('q2').value.trim() || 'life',
                seriousness: document.getElementById('q3').value,
                voice: selectedVoice
            };
            
            document.getElementById('question3').style.display = 'none';
            document.getElementById('sessionContainer').style.display = 'block';
            
            await generateInitialTherapy();
            startTimer();
            
            // Show voice switcher after 30 seconds
            setTimeout(() => {
                document.getElementById('voiceSwitcher').style.display = 'block';
            }, 30000);
            
            trackEvent('session_start', userAnswers);
        }
        
        async function generateInitialTherapy() {
            const response = await fetch('/api/generate-therapy', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(userAnswers)
            });
            
            const data = await response.json();
            
            const contentDiv = document.getElementById('sessionContent');
            contentDiv.textContent = data.therapy;
            contentDiv.className = `session-content ${selectedVoice}`;
            
            sessionStorage.setItem('lastTherapy', data.therapy);
            sessionStorage.setItem('userName', userAnswers.name);
            
            if (data.followUp) {
                setTimeout(() => {
                    showFollowUp(data.followUp);
                }, 5000);
            }
        }
        
        // ================= TIMER =================
        function startTimer() {
            sessionActive = true;
            secondsLeft = 120;
            
            sessionTimer = setInterval(() => {
                secondsLeft--;
                const mins = Math.floor(secondsLeft / 60);
                const secs = secondsLeft % 60;
                document.getElementById('timerDisplay').textContent = `${mins}:${secs.toString().padStart(2, '0')}`;
                
                if (secondsLeft === 30) {
                    document.getElementById('timerDisplay').classList.add('warning');
                    if (!upsellShown) {
                        showUpsell();
                        upsellShown = true;
                    }
                }
                
                if (secondsLeft <= 0) {
                    endSession();
                }
            }, 1000);
        }
        
        function endSession() {
            clearInterval(sessionTimer);
            sessionActive = false;
            document.getElementById('shareSection').style.display = 'block';
            document.getElementById('upgradeCard').style.display = 'block';
            generateShareSnippets();
            trackEvent('session_complete');
        }
        
        // ================= FOLLOW-UP SYSTEM =================
        function showFollowUp(question) {
            if (followUpActive) return;
            
            currentFollowUpQuestion = question;
            document.getElementById('followUpContainer').style.display = 'block';
            document.getElementById('followUpQuestion').textContent = `"${question}"`;
            
            // Style based on voice
            const card = document.getElementById('followUpCard');
            card.className = `follow-up-card ${selectedVoice}`;
            document.getElementById('followUpAvatar').textContent = voiceConfig[selectedVoice].avatar;
            document.getElementById('followUpName').textContent = `${voiceConfig[selectedVoice].name} ASKS:`;
            
            followUpActive = true;
            document.getElementById('followUpContainer').scrollIntoView({ behavior: 'smooth' });
        }
        
        async function submitFollowUp() {
            const answer = document.getElementById('followUpAnswer').value.trim();
            if (!answer) {
                alert('Please answer the question');
                return;
            }
            
            document.getElementById('followUpSubmit').disabled = true;
            document.getElementById('followUpSubmit').textContent = '⏳ THINKING...';
            
            try {
                const response = await fetch('/api/generate-therapy', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        ...userAnswers,
                        previousAnswer: answer,
                        followUpIndex: followUpIndex
                    })
                });
                
                const data = await response.json();
                
                // Store exchange
                exchanges.push({
                    question: currentFollowUpQuestion,
                    answer: answer,
                    response: data.message
                });
                
                // Show response
                const responseDiv = document.getElementById('followUpResponse');
                responseDiv.style.display = 'block';
                responseDiv.innerHTML = `
                    <div style="display: flex; align-items: center; gap: 10px; margin-bottom: 8px;">
                        <span>${voiceConfig[selectedVoice].avatar}</span>
                        <span style="color: ${voiceConfig[selectedVoice].color};">RESPONDS:</span>
                    </div>
                    <div>"${data.message}"</div>
                `;
                
                document.getElementById('followUpAnswer').value = '';
                
                if (data.nextQuestion) {
                    followUpIndex++;
                    setTimeout(() => {
                        document.getElementById('followUpResponse').style.display = 'none';
                        document.getElementById('followUpQuestion').textContent = `"${data.nextQuestion}"`;
                        document.getElementById('followUpSubmit').disabled = false;
                        document.getElementById('followUpSubmit').textContent = '📤 SEND ANSWER';
                    }, 4000);
                } else {
                    setTimeout(() => {
                        document.getElementById('followUpContainer').style.display = 'none';
                        document.getElementById('followUpSubmit').disabled = false;
                        document.getElementById('followUpSubmit').textContent = '📤 SEND ANSWER';
                        generateShareSnippets();
                    }, 4000);
                }
                
                trackEvent('follow_up_answer', { index: followUpIndex });
                
            } catch (error) {
                alert('Error sending answer');
                document.getElementById('followUpSubmit').disabled = false;
                document.getElementById('followUpSubmit').textContent = '📤 SEND ANSWER';
            }
        }
        
        function skipFollowUp() {
            if (confirm("Skip this question?")) {
                document.getElementById('followUpContainer').style.display = 'none';
                followUpActive = false;
                trackEvent('follow_up_skipped');
            }
        }
        
        // ================= VOICE SWITCHING =================
        async function switchVoice(newVoice) {
            const oldVoice = selectedVoice;
            selectedVoice = newVoice;
            
            // Update UI
            document.getElementById('timerDisplay').className = `timer-display ${newVoice}`;
            
            // Add transition message
            const contentDiv = document.getElementById('sessionContent');
            contentDiv.innerHTML += `
                <div style="margin: 15px 0; padding: 10px; background: #333; border-left: 3px solid ${voiceConfig[newVoice].color};">
                    <span style="color: ${voiceConfig[newVoice].color};">${voiceConfig[newVoice].avatar} ${voiceConfig[newVoice].name} joins the conversation...</span>
                </div>
            `;
            
            // Generate new response with new voice
            const response = await fetch('/api/generate-therapy', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    ...userAnswers,
                    voice: newVoice,
                    context: exchanges
                })
            });
            
            const data = await response.json();
            
            contentDiv.innerHTML += `
                <div class="session-content ${newVoice}" style="margin-top: 10px;">
                    "${data.therapy}"
                </div>
            `;
            
            trackEvent('voice_switched', { from: oldVoice, to: newVoice });
        }
        
        // ================= SHARE SYSTEM =================
        function generateShareSnippets() {
            const therapy = sessionStorage.getItem('lastTherapy') || '';
            const snippets = [];
            
            // Add main therapy snippet
            snippets.push({
                text: therapy.substring(0, 80) + '...',
                voice: selectedVoice
            });
            
            // Add exchange snippets
            exchanges.forEach((ex, i) => {
                snippets.push({
                    text: `I told ${voiceConfig[selectedVoice].name}: "${ex.answer.substring(0, 50)}..."`,
                    voice: selectedVoice
                });
            });
            
            let html = '';
            snippets.slice(0, 3).forEach((s, i) => {
                html += `
                    <div class="snippet-card" onclick="shareSnippet(${i})">
                        <div class="snippet-text">"${s.text}"</div>
                        <div class="snippet-voice" style="color: ${voiceConfig[s.voice].color};">${voiceConfig[s.voice].avatar} ${voiceConfig[s.voice].name}</div>
                    </div>
                `;
            });
            
            document.getElementById('snippetGrid').innerHTML = html;
        }
        
        function shareSnippet(index) {
            const snippets = [
                sessionStorage.getItem('lastTherapy'),
                ...exchanges.map(e => e.answer)
            ];
            
            const text = encodeURIComponent(`"${snippets[index].substring(0, 80)}..." Get your therapy at trumpbot.rip 🔥`);
            const url = encodeURIComponent('https://trumpbot.rip/therapy');
            
            window.open(`https://twitter.com/intent/tweet?text=${text}&url=${url}`, '_blank');
            trackEvent('share_snippet');
        }
        
        function shareFull(platform) {
            const therapy = sessionStorage.getItem('lastTherapy') || '';
            const text = encodeURIComponent(`My session with ${voiceConfig[selectedVoice].name}: "${therapy.substring(0, 100)}..." Try it at trumpbot.rip 🔥`);
            const url = encodeURIComponent('https://trumpbot.rip/therapy');
            
            if (platform === 'twitter') {
                window.open(`https://twitter.com/intent/tweet?text=${text}&url=${url}`, '_blank');
            } else if (platform === 'facebook') {
                window.open(`https://www.facebook.com/sharer/sharer.php?u=${url}&quote=${text}`, '_blank');
            } else {
                navigator.clipboard.writeText(`My therapy session: "${therapy}"\n\nGet yours: https://trumpbot.rip/therapy`);
                alert('✅ Copied!');
            }
            
            trackEvent('share_full', { platform });
        }
        
        // ================= PAYMENT =================
        function selectPlan(plan) {
            currentPlan = plan;
            document.querySelectorAll('.pricing-option').forEach(opt => opt.classList.remove('selected'));
            document.getElementById(`plan${plan.charAt(0).toUpperCase() + plan.slice(1)}`).classList.add('selected');
            
            const btn = document.getElementById('upgradeBtn');
            btn.textContent = plan === 'single' ? '🔓 UNLOCK - $2.99' : '🔓 UNLOCK WEEKLY - $9.99';
        }
        
        async function processPayment() {
            try {
                const response = await fetch('/api/create-checkout', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        plan: currentPlan,
                        metadata: {
                            name: userAnswers.name,
                            voice: selectedVoice,
                            exchanges: exchanges.length
                        }
                    })
                });
                
                const session = await response.json();
                await stripe.redirectToCheckout({ sessionId: session.id });
                
            } catch (error) {
                alert('Payment failed');
            }
        }
        
        // ================= UPSELL =================
        function showUpsell() {
            document.getElementById('upsellModal').classList.add('active');
        }
        
        function closeUpsell() {
            document.getElementById('upsellModal').classList.remove('active');
        }
        
        async function buyMoreTime() {
            closeUpsell();
            secondsLeft += 180; // Add 3 minutes
            document.getElementById('timerDisplay').classList.remove('warning');
            trackEvent('bought_more_time');
        }
        
        // ================= VIRAL TRACKING =================
        function updateLiveCounter() {
            const counts = [2847, 2956, 3123, 3345, 3567];
            setInterval(() => {
                document.getElementById('liveCounter').innerHTML = `⚡ ${counts[Math.floor(Math.random() * counts.length)].toLocaleString()} in session now ⚡`;
            }, 8000);
        }
        
        function startSocialProofUpdates() {
            setInterval(() => {
                ['totalSessions', 'sharesToday'].forEach(id => {
                    const el = document.getElementById(id);
                    el.textContent = (parseInt(el.textContent.replace(',', '')) + Math.floor(Math.random() * 50)).toLocaleString();
                });
            }, 10000);
            
            setInterval(() => {
                if (Math.random() > 0.7) {
                    const badge = document.getElementById('viralBadge');
                    badge.innerHTML = `<div class="viral-badge">🔥 ${Math.floor(Math.random() * 50) + 20} sharing now!</div>`;
                    setTimeout(() => { badge.innerHTML = ''; }, 4000);
                }
            }, 15000);
        }
        
        function trackEvent(event, data = {}) {
            fetch('/api/track-viral', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ event, sessionId, data, timestamp: Date.now() })
            }).catch(() => {});
        }
        
        function checkForPaymentReturn() {
            const params = new URLSearchParams(window.location.search);
            if (params.get('session_id')) {
                document.getElementById('sessionContainer').style.display = 'block';
                document.getElementById('sessionContent').textContent = '✅ Payment confirmed! Loading premium session...';
                setTimeout(() => generateInitialTherapy(), 2000);
                window.history.replaceState({}, document.title, '/therapy-multi.html');
            }
        }
        
        // Initialize
        window.onload = init;
    </script>
